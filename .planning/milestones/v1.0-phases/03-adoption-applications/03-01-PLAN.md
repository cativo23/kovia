```typescript
@ApiTags('Applications')
@Controller('applications')
export class ApplicationsController {
  constructor(private readonly applicationsService: ApplicationsService) {}

  @Post()
  @Roles('ADOPTER')
  async create(@Body() dto: CreateApplicationDto, @Req() req) {
    return this.applicationsService.create(dto, req.user);
  }

  @Get('check')
  async checkExisting(@Query('animalId') animalId: string, @Req() req) {
    return this.applicationsService.checkExisting(animalId, req.user.sub);
  }

  @Get('my')
  async findMyApplications(@Req() req, @Query() query: ApplicationQueryDto) {
    return this.applicationsService.findMyApplications(req.user.sub, query);
  }

  @Get('my/:id')
  async findMyApplication(@Param('id') id: string, @Req() req) {
    return this.applicationsService.findById(id, req.user.sub);
  }

  @Patch(':id/retirar')
  async withdraw(@Param('id') id: string, @Req() req) {
    return this.applicationsService.withdraw(id, req.user.sub);
  }

  @Get('org')
  @Roles('ORG_ADMIN')
  async findOrgApplications(@Query() query: ApplicationQueryDto) {
    return this.applicationsService.findAllByOrg(query);
  }

  @Get('org/:id')
  @Roles('ORG_ADMIN')
  async findOrgApplication(@Param('id') id: string) {
    return this.applicationsService.findByIdForOrg(id);
  }

  @Patch(':id/status')
  @Roles('ORG_ADMIN')
  async updateStatus(@Param('id') id: string, @Body() dto: UpdateApplicationStatusDto, @Req() req) {
    return this.applicationsService.updateStatus(id, dto.status, req.user.sub);
  }
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Adopter -> API | Untrusted form data + photo references from authenticated adopter |
| Staff -> API | Status transitions from org admin — must be scoped to own org |
| Adopter -> other adopter data | Must not be able to read another adopter's application |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Tampering | ApplicationsController POST / | mitigate | class-validator DTOs validate all input; Zod on frontend is defense-in-depth |
| T-03-02 | Information Disclosure | GET /applications/my | mitigate | publicPrisma with explicit `where: { userId: req.user.sub }` — no org RLS needed |
| T-03-03 | Information Disclosure | GET /applications/org | mitigate | prismaRls enforces org scoping via CLS `app.current_org_id` |
| T-03-04 | Elevation of Privilege | PATCH /:id/status | mitigate | @Roles('ORG_ADMIN') guard + RLS scoping — adopter cannot change status |
| T-03-05 | Tampering | PATCH /:id/retirar | mitigate | Ownership check `app.userId === req.user.sub` in service layer |
| T-03-06 | Tampering | State machine bypass | mitigate | `staffTransitions` map in service layer rejects invalid transitions with BadRequestException |
| T-03-07 | Repudiation | All status changes | mitigate | auditService.log() on every status change with old/new status |
</threat_model>
